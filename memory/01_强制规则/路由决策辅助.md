# Agent 路由决策辅助函数

## 使用方法
在派发任务前调用 `get_best_agent_for_task()` 获取推荐 agent。

## 函数

```python
import json
from pathlib import Path

def get_provider_status():
    """读取 provider 状态"""
    status_file = Path("/Users/apple/.openclaw/workspace/memory/ops/provider-status.json")
    if not status_file.exists():
        return {}
    return json.loads(status_file.read_text())

def is_model_available(model_full_name):
    """检查模型是否可用
    
    Args:
        model_full_name: 如 "mynewapi/claude-opus-4-6"
    
    Returns:
        bool: True=可用, False=不可用
    """
    status = get_provider_status()
    if not status or "providers" not in status:
        return True  # 状态未知时默认可用
    
    provider, model = model_full_name.split("/")
    
    if provider not in status["providers"]:
        return True
    
    models = status["providers"][provider].get("models", {})
    if model not in models:
        return True
    
    model_status = models[model].get("status", "unknown")
    return model_status in ["ok", "degraded"]  # degraded 也算可用，只是降级

def get_best_agent_for_task(task_type, content_hints=None):
    """根据任务类型和 provider 状态推荐最佳 agent
    
    Args:
        task_type: "multimodal" | "code" | "chinese" | "reasoning" | "light"
        content_hints: 可选的内容提示（如关键词）
    
    Returns:
        str: 推荐的 agent id
    """
    # Agent 编制（从 openclaw.json 读取）
    agents = {
        "jimmy": "mynewapi/claude-sonnet-4-6",
        "main": "google-gemini-cli/gemini-3-pro-preview",
        "deep": "mygptapi/gpt-5.4",
        "kitt": "mynewapi/claude-opus-4-6"
    }
    
    # 任务类型 → 首选 agent
    preferences = {
        "multimodal": ["main"],  # 图片/视频/长文档
        "code": ["deep", "kitt"],  # 代码/算法
        "chinese": ["deep"],  # 中文内容
        "reasoning": ["kitt"],  # 深度推理/架构决策
        "light": ["jimmy", "main"]  # 轻量任务
    }
    
    candidates = preferences.get(task_type, ["deep"])
    
    # 按优先级检查可用性
    for agent_id in candidates:
        model = agents.get(agent_id)
        if model and is_model_available(model):
            return agent_id
    
    # 所有首选都不可用，fallback 到 main（免费且稳定）
    return "main"

# 使用示例
if __name__ == "__main__":
    print("=== Provider Status ===")
    status = get_provider_status()
    for provider, info in status.get("providers", {}).items():
        for model, details in info.get("models", {}).items():
            print(f"{provider}/{model}: {details.get('status', 'unknown')}")
    
    print("\n=== Routing Recommendations ===")
    for task_type in ["multimodal", "code", "chinese", "reasoning", "light"]:
        agent = get_best_agent_for_task(task_type)
        print(f"{task_type:12} → {agent}")
```

## 集成到 sessions_spawn

在调用 `sessions_spawn` 前：

```python
# 自动选择最佳 agent
task_type = "chinese"  # 根据任务内容判断
recommended_agent = get_best_agent_for_task(task_type)

sessions_spawn(
    agentId=recommended_agent,
    task="..."
)
```

## 状态文件格式

`ops/provider-status.json`:
```json
{
  "timestamp": "2026-03-10T11:51:30Z",
  "providers": {
    "mynewapi": {
      "models": {
        "claude-opus-4-6": {
          "node": "cc7-opus",
          "status": "ok"
        }
      }
    }
  },
  "recommendations": {
    "avoid": ["mygptapi/gpt-5.4"],
    "prefer": ["mynewapi/claude-opus-4-6"]
  }
}
```

## 状态值

- `ok`: 正常，优先使用
- `degraded`: 降级，可用但不推荐
- `down`: 不可用，路由时跳过
- `unknown`: 未知，默认可用
